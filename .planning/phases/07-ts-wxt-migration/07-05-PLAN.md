<task type="auto">
  <name>Task 1: Add tsup configuration with path alias resolution</name>
  <files>packages/mcp/tsup.config.ts</files>
  <action>
    Create tsup.config.ts to resolve @/ aliases during build:
    
    ```typescript
    import { defineConfig } from 'tsup';
    import path from 'path';
    
    export default defineConfig({
      entry: ['src/index.ts'],
      format: ['esm'],
      outDir: 'dist',
      clean: true,
      esbuildOptions(options) {
        options.alias = {
          '@': path.resolve(__dirname, 'src')
        };
      }
    });
    ```
    
    This tells esbuild (used by tsup) to resolve `@/` imports to the src/ directory.
  </action>
