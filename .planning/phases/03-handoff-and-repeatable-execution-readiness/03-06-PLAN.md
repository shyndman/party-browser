    ```javascript
      const element = document.activeElement || document.body;
      
      const event = new KeyboardEvent('keydown', {
        key,
        bubbles: true,
        cancelable: true
      });
      element.dispatchEvent(event);
      
      const upEvent = new KeyboardEvent('keyup', {
        key,
        bubbles: true,
        cancelable: true
      });
      element.dispatchEvent(upEvent);
      
      return { success: true };
    }
    ```

    ```javascript
    async function handleWaitFor(params) {
      const { mode, value, timeout = 30 } = params;
      // NOTE: Unit conversion (seconds→milliseconds) will be added in 03-07
      // Currently timeout is in seconds but Date.now() returns milliseconds
      const startTime = Date.now();
      
      if (mode === 'time') {
        // NOTE: Conversion from seconds to milliseconds will be added in 03-07
        await new Promise(resolve => setTimeout(resolve, value));
        return { success: true };
      }
      
      if (mode === 'text') {
        // NOTE: Timeout comparison unit fix will be added in 03-07
        while (Date.now() - startTime < timeout) {
          if (document.body.innerText.includes(value)) {
            return { success: true };
          }
          await new Promise(resolve => setTimeout(resolve, 100));
        }
        throw new Error(`Timeout waiting for text: "${value}"`);
      }
      
      if (mode === 'text_gone') {
        // NOTE: Timeout comparison unit fix will be added in 03-07
        while (Date.now() - startTime < timeout) {
          if (!document.body.innerText.includes(value)) {
            return { success: true };
          }
          await new Promise(resolve => setTimeout(resolve, 100));
        }
        throw new Error(`Timeout waiting for text to disappear: "${value}"`);
      }
      
      throw new Error(`Unknown wait mode: ${mode}`);
    }
    ```

    3. handleDrag:
    ```javascript
    async function handleDrag(params) {
      const { startElement, endElement } = params;
      
      const startEl = findElement(startElement);
      const endEl = findElement(endElement);
      
      if (!startEl) throw new Error(`Start element not found: ${startElement}`);
      if (!endEl) throw new Error(`End element not found: ${endElement}`);
      
      const startRect = startEl.getBoundingClientRect();
      const endRect = endEl.getBoundingClientRect();
      
      const startX = startRect.left + startRect.width / 2;
      const startY = startRect.top + startRect.height / 2;
      const endX = endRect.left + endRect.width / 2;
      const endY = endRect.top + endRect.height / 2;
      
      // Simulate drag with mouse events
      startEl.dispatchEvent(new MouseEvent('mousedown', {
        bubbles: true,
        clientX: startX,
        clientY: startY
      }));
      
      await new Promise(resolve => setTimeout(resolve, 50));
      
      document.dispatchEvent(new MouseEvent('mousemove', {
        bubbles: true,
        clientX: endX,
        clientY: endY
      }));
      
      await new Promise(resolve => setTimeout(resolve, 50));
      
      endEl.dispatchEvent(new MouseEvent('mouseup', {
        bubbles: true,
        clientX: endX,
        clientY: endY
      }));
      
      return { success: true };
    }
    ```
