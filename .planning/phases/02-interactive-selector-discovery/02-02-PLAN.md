<task type="auto">
  <name>Task 2: Add error handling for invalid screenshot data</name>
  <files>src/tools/custom.ts</files>
  <action>
    Enhance the screenshot tool with error handling for malformed or missing data from the extension.
    
    Update src/tools/custom.ts screenshot handler to:
    
    ```typescript
    export const screenshot: Tool = {
      schema: {
        name: ScreenshotTool.shape.name.value,
        description: ScreenshotTool.shape.description.value,
        inputSchema: zodToJsonSchema(ScreenshotTool.shape.arguments),
      },
      handle: async (context, _params) => {
        try {
          // Extension contract: returns base64-encoded PNG string (not Data URL)
          const screenshot = await context.sendSocketMessage(
            "browser_screenshot",
            {},
          );
          
          // Validate we received data
          if (!screenshot || typeof screenshot !== "string") {
            return {
              content: [{
                type: "text",
                text: "Error: Screenshot returned no data. Ensure the browser extension has permission to capture the active tab.",
              }],
              isError: true,
            };
          }
          
          // Strip data URL prefix if present (extension should not send this, but handle gracefully)
          let imageData = screenshot;
          if (screenshot.startsWith("data:image/png;base64,")) {
            imageData = screenshot.slice("data:image/png;base64,".length);
          } else if (screenshot.startsWith("data:")) {
            return {
              content: [{
                type: "text",
                text: "Error: Screenshot returned unexpected data URL format. Expected raw base64 PNG data.",
              }],
              isError: true,
            };
          }
          
          return {
            content: [
              {
                type: "image",
                data: imageData,
                mimeType: "image/png",
              },
            ],
          };
        } catch (error) {
          return {
            content: [{
              type: "text",
              text: `Screenshot failed: ${error instanceof Error ? error.message : String(error)}`,
            }],
            isError: true,
          };
        }
      },
    };
    ```
  </action>
  <verify>grep -A 40 "export const screenshot" src/tools/custom.ts | grep -E "(try|catch|Error|isError)"</verify>
  <done>Screenshot tool has proper error handling for missing data, Data URL stripping, and try-catch wrapper</done>
</task>
