<task type="auto">
  <name>Task 3: Create evaluate.ts tool handler</name>
  <files>src/tools/evaluate.ts</files>
  <action>
    Create a new file src/tools/evaluate.ts with the evaluate tool handler:
    
    ```typescript
    import { zodToJsonSchema } from "zod-to-json-schema";
    
    import { EvaluateTool } from "@/protocol/tool-definitions";
    
    import type { Tool } from "./tool";
    
    export const evaluate: Tool = {
      schema: {
        name: EvaluateTool.shape.name.value,
        description: EvaluateTool.shape.description.value,
        inputSchema: zodToJsonSchema(EvaluateTool.shape.arguments),
      },
      handle: async (context, params) => {
        const { script } = EvaluateTool.shape.arguments.parse(params);
        const result = await context.sendSocketMessage("browser_evaluate", { script });
        
        // Format result based on type for readable output
        let resultText: string;
        if (result === undefined) {
          resultText = "undefined";
        } else if (result === null) {
          resultText = "null";
        } else if (typeof result === "object") {
          resultText = JSON.stringify(result, null, 2);
        } else {
          resultText = String(result);
        }

        return {
          content: [{
            type: "text",
            text: `\`\`\`\n${resultText}\n\`\`\``,
          }],
        };
      },
    };
    ```
    
    This is a read-only inspection tool (no page modification), so it does NOT include a snapshot in the response - just the evaluation result formatted as text with code block.
  </action>
  <verify>cat src/tools/evaluate.ts | head -50</verify>
  <done>evaluate.ts exists with proper Tool interface implementation, sends browser_evaluate message, formats result appropriately</done>
</task>
